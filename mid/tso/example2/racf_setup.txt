/* STEP 1: DEFINE THE CORE APP FACILITY SECURITY PROFILE */
RDEFINE FACILITY SECURE.PAYLOAD UACC(NONE) AUDIT(ALL(READ))

/* STEP 2: GRANT ACCESS TO THE EXECUTION ID (REPLACE TESTUSER) */
PERMIT SECURE.PAYLOAD CLASS(FACILITY) ID(TESTUSER) ACCESS(UPDATE)

/* STEP 3: REFRESH IN-STORAGE FACILITY PROFILES */
SETROPTS RACLIST(FACILITY) REFRESH

/* STEP 4: ACCESS DEFINITIONS FOR ICSF CSNBKGN SERVICE */
RDEFINE CSFSERV CSNBKGN UACC(NONE)
PERMIT CSNBKGN CLASS(CSFSERV) ID(TESTUSER) ACCESS(READ)

/* STEP 5: REFRESH IN-STORAGE CSFSERV PROFILES */
SETROPTS RACLIST(CSFSERV) REFRESH
